    <script>
      // Modified to receive parameters dynamically from the clicked card
      // ADD 'isClothing = true' here
      function addToBag(id, name, price, image, isClothing = true) {
        const selectedProduct = {
          id: id,
          name: name,
          price: parseFloat(price),
          image: image,
          isClothing: isClothing,

          // NEW: Add variant options
          variants: isClothing
            ? {
                type: "size",
                label: "Select Size",
                options: ["S", "M", "L", "XL"],
              }
            : {
                type: "size",
                label: "Select Size",
                options: ["One Size"], // Change this for Touchpad later
              },
        };

        localStorage.setItem(
          "selectedProduct",
          JSON.stringify(selectedProduct),
        );
        window.location.href = "addToCart.html";
      }

      // Calculates total units when the page loads so the layout displays accurately
      function updateBagCount() {
        let cart = JSON.parse(localStorage.getItem("cart")) || [];
        let totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
        const bagCountBadge = document.getElementById("bag-count");

        if (totalItems > 0) {
          bagCountBadge.textContent = totalItems;
          bagCountBadge.classList.remove("hidden");
        } else {
          bagCountBadge.classList.add("hidden");
        }
      }

      // Run the item sync calculation on page mount
      updateBagCount();
    </script>
